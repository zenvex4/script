-- Reliable Server Hopping Function with API Check & Queue Rotation
local function HopServer()
    if State.IsHopping then return end
    State.IsHopping = true
    StatusLabel.Text, StatusLabel.TextColor3 = "Finding Server...", Color3.fromRGB(180, 120, 255)
    
    task.spawn(function()
        local httpreq = (request or http_request or syn and syn.request)
        local vServers = {}
        local cursor = ""
        local successFound = false
        
        for i = 1, 3 do
            local url = string.format("https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=Asc&limit=100%s", tostring(PlaceId), cursor ~= "" and "&cursor=" .. cursor or "")
            
            local s, r = pcall(function()
                if httpreq then 
                    local res = httpreq({Url=url, Method="GET"})
                    return HttpService:JSONDecode(res.Body)
                else 
                    return HttpService:JSONDecode(game:HttpGet(url)) 
                end
            end)

            if s and r and r.data then
                for _, srv in ipairs(r.data) do
                    if type(srv) == "table" and srv.id ~= JobId and srv.playing and srv.playing >= Config.MinPlayers and srv.playing <= Config.MaxPlayers then
                        table.insert(vServers, srv.id)
                    end
                end
                
                if r.nextPageCursor and #vServers == 0 then
                    cursor = r.nextPageCursor
                else
                    break
                end
            else
                break
            end
        end

        if #vServers > 0 then
            State.TotalHops = State.TotalHops + 1
            StatusLabel.Text = "Teleporting to new server..."
            
            -- Shuffle to make sure it doesn't always pick index 1
            local targetServerId = vServers[math.random(1, #vServers)]
            
            local tpSuccess = false
            pcall(function()
                TeleportService:TeleportToPlaceInstance(PlaceId, targetServerId, LocalPlayer)
                tpSuccess = true
            end)
            
            if tpSuccess then
                task.wait(8)
                State.IsHopping = false
                return
            end
        end
        
        -- If HTTP request failed completely or instance teleport failed, 
        -- use a Teleport queue fallback hook to force an alternate server pool join
        State.TotalHops = State.TotalHops + 1
        StatusLabel.Text = "Using alternative hop..."
        
        local ts = game:GetService("TeleportService")
        pcall(function()
            -- Teleport to a fresh server pool instance using reserved/alternative handles if supported, or standard teleport with a delay trick
            ts:Teleport(PlaceId, LocalPlayer)
        end)
        
        task.wait(5)
        State.IsHopping = false
    end)
end

local function ForceClickAutoFarm(btnTarget)
    if not btnTarget then return false end
    local success = false
    pcall(function()
        if firesignal and btnTarget:IsA("GuiButton") then
            pcall(function() firesignal(btnTarget.Activated) end)
            success = true
        elseif getconnections then
            for _, conn in pairs(getconnections(btnTarget.Activated)) do 
                pcall(function() conn:Fire() end) 
                success = true 
                break 
            end
            if not success then
                for _, conn in pairs(getconnections(btnTarget.MouseButton1Click)) do 
                    pcall(function() conn:Fire() end) 
                    success = true 
                    break 
                end
            end
        end
        
        if not success then
            local inset = GuiService:GetGuiInset()
            local cx = btnTarget.AbsolutePosition.X + (btnTarget.AbsoluteSize.X / 2) + inset.X
            local cy = btnTarget.AbsolutePosition.Y + (btnTarget.AbsoluteSize.Y / 2) + inset.Y
            pcall(function() VirtualInputManager:SendTouchEvent(1, 0, cx, cy) end)
            task.wait(0.05)
            pcall(function() VirtualInputManager:SendTouchEvent(1, 1, cx, cy) end)
        end
    end)
    return success
end

local function AnalyzeGameText(txt)
    if type(txt) ~= "string" then return end
    local t = string.lower(txt)
    if string.find(t, "auto") and (string.find(t, "mine") or string.find(t, "farm")) then
        if string.find(t, "enabled") or string.find(t, "on") then
            State.IsAutoFarmOn = true
        elseif string.find(t, "disabled") or string.find(t, "off") then
            State.IsAutoFarmOn = false
        end
    end
end

local function CheckBlock(b)
    if b:GetAttribute("Broken") or b:GetAttribute("Destroyed") then return false end
    local hp = b:GetAttribute("Health") or b:GetAttribute("health") or b:GetAttribute("HP")
    if hp ~= nil and tonumber(hp) <= 0 then return false end

    local nameStr = string.lower(b.Name):gsub("[%s_%-]", "")
    if string.find(nameStr, "onyx") or Config.TargetIDs[nameStr] then return true end

    local isOnyx = false
    for _, key in ipairs({"id", "Id", "ID", "Type"}) do
        local val = b:GetAttribute(key)
        if val then
            local strVal = string.lower(tostring(val)):gsub("[%s_%-]", "")
            if string.find(strVal, "onyx") or Config.TargetIDs[strVal] then 
                isOnyx = true; break 
            end
        end
    end

    if not isOnyx then return false end
    if (b:IsA("Model") or b:IsA("Folder")) and #b:GetChildren() == 0 then return false end
    return true
end

local function UpdateMasterScanner()
    local count = 0
    local firstPart, firstId = nil, nil
    local things = Workspace:FindFirstChild("__THINGS")
    
    if things then
        State.InMiningZone = true
        local function scanFolder(f)
            for _, b in ipairs(f:GetDescendants()) do
                if CheckBlock(b) then
                    count = count + 1
                    if not firstPart then
                        firstPart = b:IsA("BasePart") and b or b:FindFirstChildWhichIsA("BasePart", true)
                        firstId = b.Name
                    end
                end
            end
        end
        local bw = things:FindFirstChild("BlockWorlds")
        if bw then scanFolder(bw) end
        local br = things:FindFirstChild("Breakables")
        if br then
            scanFolder(br)
        end
    else 
        State.InMiningZone = false 
    end
    
    State.OnyxCount = count
    State.TargetPart = firstPart
    State.TargetId = firstId
end
